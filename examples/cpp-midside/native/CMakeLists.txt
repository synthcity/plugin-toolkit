cmake_minimum_required(VERSION 3.25)
project(PortableMidSide VERSION 0.1.0 LANGUAGES C CXX)
if(APPLE)
  enable_language(OBJC OBJCXX)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
set(CLAP_WRAPPER_DOWNLOAD_DEPENDENCIES ON CACHE BOOL "Fetch wrapper dependencies")
set(CLAP_WRAPPER_COPY_AFTER_BUILD OFF CACHE BOOL "Keep qualification builds local" FORCE)
include(FetchContent)
FetchContent_Declare(clap_wrapper GIT_REPOSITORY https://github.com/free-audio/clap-wrapper.git
  GIT_TAG 1cca996e96f29ab2be7ae9f8cfe532bbc92e1dd6)
FetchContent_MakeAvailable(clap_wrapper)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(EXAMPLE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
execute_process(COMMAND ${Python3_EXECUTABLE} "${EXAMPLE_ROOT}/prepare.py" COMMAND_ERROR_IS_FATAL ANY)
add_library(midside_impl STATIC "${EXAMPLE_ROOT}/src/webclap.cpp")
target_include_directories(midside_impl PRIVATE "${EXAMPLE_ROOT}/include" "${EXAMPLE_ROOT}/build")
# Only the entry symbol differs. The DSP and CLAP implementation are unchanged.
target_compile_definitions(midside_impl PRIVATE clap_entry=portable_midside_entry)
target_link_libraries(midside_impl PUBLIC clap clap-wrapper-extensions)
set(MIDSIDE_FORMATS "CLAP;VST3;AUV2" CACHE STRING "Local qualification formats")
make_clapfirst_plugins(TARGET_NAME portable_midside IMPL_TARGET midside_impl
  ENTRY_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/entry.cpp"
  OUTPUT_NAME "Portable MidSide" BUNDLE_IDENTIFIER "org.synthcity.examples.hetrick-midside"
  BUNDLE_VERSION ${PROJECT_VERSION} COPY_AFTER_BUILD FALSE
  PLUGIN_FORMATS ${MIDSIDE_FORMATS} ASSET_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugins"
  AUV2_MANUFACTURER_NAME "SynthCity" AUV2_MANUFACTURER_CODE "SyCi"
  AUV2_SUBTYPE_CODE "HtMS" AUV2_INSTRUMENT_TYPE "aufx")

enable_testing()
find_program(CLAP_VALIDATOR clap-validator DOC "Optional upstream CLAP validator")
find_program(VST3_VALIDATOR vst3-validator DOC "Optional Steinberg VST3 SDK validator")
if(CLAP_VALIDATOR)
  if(APPLE)
    set(CLAP_TEST_PATH "$<TARGET_BUNDLE_DIR:portable_midside_clap>")
  else()
    set(CLAP_TEST_PATH "$<TARGET_FILE:portable_midside_clap>")
  endif()
  add_test(NAME clap-validator COMMAND "${CLAP_VALIDATOR}" -v quiet validate --json "${CLAP_TEST_PATH}")
endif()
if(VST3_VALIDATOR AND TARGET portable_midside_vst3)
  if(APPLE)
    set(VST3_TEST_PATH "$<TARGET_BUNDLE_DIR:portable_midside_vst3>")
  else()
    set(VST3_TEST_PATH "$<TARGET_FILE:portable_midside_vst3>")
  endif()
  add_test(NAME vst3-validator COMMAND "${VST3_VALIDATOR}" "${VST3_TEST_PATH}")
endif()
if(APPLE AND TARGET portable_midside_auv2)
  add_executable(au-host "${EXAMPLE_ROOT}/tests/au-host.cpp")
  target_compile_options(au-host PRIVATE -UNDEBUG)
  target_link_libraries(au-host PRIVATE "-framework AudioToolbox" "-framework CoreFoundation")
  add_dependencies(au-host portable_midside_auv2)
  add_test(NAME au-numerical-state COMMAND au-host "$<TARGET_FILE:portable_midside_auv2>")
endif()
